def embed_summary(key, text):
    return {
        "file": key,
        "lines": text.count("\n") + (1 if text else 0),
        "bytes": len(text.encode("utf-8")),
        "sha256": hashlib.sha256(text.encode("utf-8")).hexdigest()[:16] if text else None,
    }
